+    if (argc != 3) return picolArityErr(i, argv[0]);
+    char *s = argv[2];
+    char *p = s + strlen(s);
+    /* Skip trailing slashes */
+    while (p > s && *(p-1) == '/') p--;
+    /* Find last / */
+    while (p > s && *--p != '/') ;
+    if (p == s && *p == '/') {
+        /* Root directory: dirname of / is / */
+        picolSetResult(i,"/");
+    } else if (p == s) {
+        /* No / found, return "." */
