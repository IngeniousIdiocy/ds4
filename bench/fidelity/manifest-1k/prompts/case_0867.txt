+/* Helper: read file into xmalloc'd buffer, returns NULL on error. */
+static char *picolReadFile(const char *filename, size_t *len_out) {
+    FILE *fp = fopen(filename,"rb");
+    if (!fp) return NULL;
+    size_t cap = 4096, len = 0;
+    char *buf = xmalloc(cap);
+    size_t n;
+    while ((n = fread(buf+len, 1, cap-len, fp)) > 0) {
+        len += n;
+        if (cap - len < 1024) {
+            if (cap > (SIZE_MAX / 2)) { /* overflow check */
+                fclose(fp);
+                free(buf);
+                return NULL;
+            }
+            cap *= 2;
+            buf = xrealloc(buf, cap);
+        }
+    }
+    fclose(fp);
+    buf[len] = '\0';
+    if (len_out) *len_out = len;
+    return buf;
+}
